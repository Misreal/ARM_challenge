# Raspberry Pi 5 benchmark agent dependencies. Deliberately minimal.
#
# Install on a 64-bit Raspberry Pi OS (Bookworm). A 32-bit OS has no aarch64
# onnxruntime wheel and is not supported by this project.
#
#   python3 -m venv ~/armopt && source ~/armopt/bin/activate
#   pip install -r requirements-pi.txt
#
# NEVER install torch or torchvision here. The agent loads ONNX only; a torch
# install on the Pi invites accidental torch-vs-ORT comparisons that would not
# reflect the deployment target.
#
# The onnxruntime version MUST match the PC exactly for the whole campaign --
# kernel changes between releases shift both accuracy and latency. Verified
# 2026-07-28: cp311 manylinux_2_28 aarch64 wheels exist for both pins below, so
# Bookworm's system Python 3.11 (glibc 2.36) installs them without a source
# build. Do not relax these pins mid-campaign.
#
# onnx is not optional despite the agent "loading ONNX only": the device entry
# points (src.quant.measure / baselines / saturation_probe) import
# onnxruntime.quantization and src.quant.groups, both of which import onnx
# directly. Omitting it fails at import, not at runtime.

numpy>=1.26
onnx==1.22.0
onnxruntime==1.27.0
psutil>=5.9
